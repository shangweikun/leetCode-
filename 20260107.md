class Solution {
    public int findShortestSubArray(int[] nums) {

        if (nums.length == 1) {
            return 1;
        }

        int max;
        int result;
        Map<Integer, List<Integer>> m = new HashMap<>();

        List<Integer> r = new ArrayList<>();
        r.add(0);
        r.add(0);
        r.add(1);
        r.add(0-0+1);
        m.put(nums[0], r);
        max = 1;
        result = nums[0];

        for (int i = 1; i < nums.length; i++) {
            r = m.get(nums[i]);
            if (r == null) {
                r = new ArrayList<>();
                r.add(i);
                r.add(i);
                r.add(1);
                r.add(i - i +1);
                m.put(nums[i], r);
            } else {
                List<Integer> r1 = new ArrayList<>();
                r1.add(r.get(0));
                r1.add(i);
                r1.add(r.get(2) + 1);
                r1.add( i - r1.get(0) + 1);
                if(r1.get(2) > max){
                    result = nums[i];
                    max = r1.get(2);
                }else if (r1.get(2) == max && r1.get(3) < m.get(result).get(3)){
                    result = nums[i];
                }

                m.put(nums[i], r1);
            }
        }

        r = m.get(result);
        return r.get(1) - r.get(0) + 1;
    }
}
