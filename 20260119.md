class Solution {

    public List<Integer> getRow(int rowIndex) {
        return test(rowIndex);
    }

    public List<Integer> test(int rowIndex){
        if(rowIndex == 0){
            List<Integer> list = new ArrayList<>();
            list.add(1);
            return list;
        }else{
            List<Integer> list = test(rowIndex -1);
            List<Integer> list1 = new ArrayList<>();
            list1.add(1);
            for(int i = 1;i<list.size();i++){
                list1.add(list.get(i-1)+list.get(i));
            }
            list1.add(1);
            return list1;
        }
    }
}

class Solution {
    public List<Integer> getRow(int rowIndex) {
        List<List<Integer>> C = new ArrayList<List<Integer>>();
        for (int i = 0; i <= rowIndex; ++i) {
            List<Integer> row = new ArrayList<Integer>();
            for (int j = 0; j <= i; ++j) {
                if (j == 0 || j == i) {
                    row.add(1);
                } else {
                    row.add(C.get(i - 1).get(j - 1) + C.get(i - 1).get(j));
                }
            }
            C.add(row);
        }
        return C.get(rowIndex);
    }
}

int* getRow(int rowIndex, int* returnSize) {
    *returnSize = rowIndex + 1;
    int* pre = malloc(sizeof(int) * (*returnSize));
    memset(pre, 0, sizeof(int) * (*returnSize));
    int* cur = malloc(sizeof(int) * (*returnSize));
    memset(cur, 0, sizeof(int) * (*returnSize));
    for (int i = 0; i <= rowIndex; i++) {
        int* tmp = cur;
        cur = pre;
        pre = tmp;
        cur[0] = cur[i] = 1;
        for (int j = 1; j < i; j++) {
            cur[j] = pre[j - 1] + pre[j];
        }
    }
    return cur;
}
